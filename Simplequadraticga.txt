/* ================================================================
   GENETIC ALGORITHM
   GLOBAL MINIMUM + GLOBAL MAXIMUM
   APDE / PROCESSING FOR ANDROID

   OBJECTIVE:
       f(x) = x(10-x)

   DOMAIN:
       0 <= x <= 10

   THEORETICAL:
       GLOBAL MINIMUM = 0
       at x = 0 or x = 10

       GLOBAL MAXIMUM = 25
       at x = 5

   TWO INDEPENDENT GENETIC ALGORITHMS:

       GA 1 -> MINIMIZATION
       GA 2 -> MAXIMIZATION

   TOTAL:
       1000 ITERATIONS
================================================================ */


// ================================================================
// GA PARAMETERS
// ================================================================

final int POP_SIZE = 32;
final int MAX_ITERATIONS = 1000;

final float X_MIN = 0.0;
final float X_MAX = 10.0;

final float CROSSOVER_RATE = 0.85;
final float MUTATION_RATE = 0.20;
final float MUTATION_SIGMA = 0.45;

final int TOURNAMENT_SIZE = 3;


// ================================================================
// MINIMUM GA
// ================================================================

float[] minPopulation = new float[POP_SIZE];
float[] minFitness = new float[POP_SIZE];
float[] minNextPopulation = new float[POP_SIZE];


// ================================================================
// MAXIMUM GA
// ================================================================

float[] maxPopulation = new float[POP_SIZE];
float[] maxFitness = new float[POP_SIZE];
float[] maxNextPopulation = new float[POP_SIZE];


// ================================================================
// GLOBAL MINIMUM
// ================================================================

float globalMinX;
float globalMinValue;
int globalMinIteration;


// ================================================================
// GLOBAL MAXIMUM
// ================================================================

float globalMaxX;
float globalMaxValue;
int globalMaxIteration;


// ================================================================
// ITERATION CONTROL
// ================================================================

int iteration = 0;

boolean running = true;
boolean finished = false;

int iterationDelay = 35;
int lastIterationTime;


// ================================================================
// HISTORY
// ================================================================

float[] minHistory = new float[MAX_ITERATIONS + 1];
float[] maxHistory = new float[MAX_ITERATIONS + 1];


// ================================================================
// GRAPH VARIABLES
// ================================================================

float graphLeft;
float graphRight;
float graphTop;
float graphBottom;

float convergenceTop;
float convergenceBottom;


// ================================================================
// SETTINGS
// ================================================================

void settings() {

  // IMPORTANT FOR APDE ANDROID
  fullScreen(P3D);
}


// ================================================================
// SETUP
// ================================================================

void setup() {

  // Android landscape
  orientation(LANDSCAPE);

  // Avoid excessive smoothing problems on some devices
  smooth();

  textAlign(LEFT, TOP);

  calculateLayout();

  resetGA();
}


// ================================================================
// RESPONSIVE LAYOUT
// ================================================================

void calculateLayout() {

  graphLeft = width * 0.055;
  graphRight = width * 0.965;

  graphTop = height * 0.145;
  graphBottom = height * 0.525;

  convergenceTop = height * 0.665;
  convergenceBottom = height * 0.855;
}


// ================================================================
// DRAW
// ================================================================

void draw() {

  background(7, 13, 23);

  // --------------------------------------------------------------
  // AUTOMATIC EVOLUTION
  // --------------------------------------------------------------

  if (running && !finished) {

    if (millis() - lastIterationTime >= iterationDelay) {

      evolveOneIteration();

      lastIterationTime = millis();
    }
  }


  // --------------------------------------------------------------
  // DRAW UI
  // --------------------------------------------------------------

  drawHeader();

  drawFunctionGraph();

  drawConvergenceGraph();

  drawStatus();

  drawButtons();
}


// ================================================================
// OBJECTIVE FUNCTION
// ================================================================

float objective(float x) {

  return x * (10.0 - x);
}


// ================================================================
// RESET
// ================================================================

void resetGA() {

  iteration = 0;

  running = true;
  finished = false;


  // --------------------------------------------------------------
  // RESET GLOBAL MINIMUM
  // --------------------------------------------------------------

  globalMinX = 0;

  globalMinValue = Float.MAX_VALUE;

  globalMinIteration = 0;


  // --------------------------------------------------------------
  // RESET GLOBAL MAXIMUM
  // --------------------------------------------------------------

  globalMaxX = 0;

  globalMaxValue = -Float.MAX_VALUE;

  globalMaxIteration = 0;


  // --------------------------------------------------------------
  // RESET HISTORY
  // --------------------------------------------------------------

  for (int i = 0; i <= MAX_ITERATIONS; i++) {

    minHistory[i] = Float.MAX_VALUE;

    maxHistory[i] = -Float.MAX_VALUE;
  }


  // --------------------------------------------------------------
  // CREATE POPULATIONS
  // --------------------------------------------------------------

  initializeMinimumPopulation();

  initializeMaximumPopulation();


  // --------------------------------------------------------------
  // EVALUATE INITIAL POPULATIONS
  // --------------------------------------------------------------

  evaluateMinimumPopulation();

  evaluateMaximumPopulation();


  // --------------------------------------------------------------
  // SAVE INITIAL VALUES
  // --------------------------------------------------------------

  minHistory[0] = globalMinValue;

  maxHistory[0] = globalMaxValue;


  lastIterationTime = millis();
}


// ================================================================
// INITIALIZE MINIMUM POPULATION
// ================================================================

void initializeMinimumPopulation() {

  for (int i = 0; i < POP_SIZE; i++) {

    minPopulation[i] =
      random(X_MIN, X_MAX);

    minFitness[i] =
      objective(minPopulation[i]);
  }
}


// ================================================================
// INITIALIZE MAXIMUM POPULATION
// ================================================================

void initializeMaximumPopulation() {

  for (int i = 0; i < POP_SIZE; i++) {

    maxPopulation[i] =
      random(X_MIN, X_MAX);

    maxFitness[i] =
      objective(maxPopulation[i]);
  }
}


// ================================================================
// EVALUATE MINIMUM
// ================================================================

void evaluateMinimumPopulation() {

  for (int i = 0; i < POP_SIZE; i++) {

    minFitness[i] =
      objective(minPopulation[i]);

    if (minFitness[i] < globalMinValue) {

      globalMinValue =
        minFitness[i];

      globalMinX =
        minPopulation[i];

      globalMinIteration =
        iteration;
    }
  }
}


// ================================================================
// EVALUATE MAXIMUM
// ================================================================

void evaluateMaximumPopulation() {

  for (int i = 0; i < POP_SIZE; i++) {

    maxFitness[i] =
      objective(maxPopulation[i]);

    if (maxFitness[i] > globalMaxValue) {

      globalMaxValue =
        maxFitness[i];

      globalMaxX =
        maxPopulation[i];

      globalMaxIteration =
        iteration;
    }
  }
}


// ================================================================
// ONE ITERATION
// ================================================================

void evolveOneIteration() {

  if (iteration >= MAX_ITERATIONS) {

    finished = true;
    running = false;

    return;
  }


  // --------------------------------------------------------------
  // MINIMIZATION
  // --------------------------------------------------------------

  evolveMinimumPopulation();


  // --------------------------------------------------------------
  // MAXIMIZATION
  // --------------------------------------------------------------

  evolveMaximumPopulation();


  // --------------------------------------------------------------
  // ADVANCE
  // --------------------------------------------------------------

  iteration++;


  // --------------------------------------------------------------
  // EVALUATE
  // --------------------------------------------------------------

  evaluateMinimumPopulation();

  evaluateMaximumPopulation();


  // --------------------------------------------------------------
  // SAVE HISTORY
  // --------------------------------------------------------------

  minHistory[iteration] =
    globalMinValue;

  maxHistory[iteration] =
    globalMaxValue;


  // --------------------------------------------------------------
  // FINISHED
  // --------------------------------------------------------------

  if (iteration >= MAX_ITERATIONS) {

    finished = true;
    running = false;
  }
}


// ================================================================
// EVOLVE MINIMUM POPULATION
// ================================================================

void evolveMinimumPopulation() {

  // --------------------------------------------------------------
  // ELITISM
  // --------------------------------------------------------------

  int eliteIndex =
    findBestMinimumIndex();

  minNextPopulation[0] =
    minPopulation[eliteIndex];


  // --------------------------------------------------------------
  // CHILDREN
  // --------------------------------------------------------------

  for (int i = 1; i < POP_SIZE; i++) {

    float parentA =
      minimumTournamentSelection();

    float parentB =
      minimumTournamentSelection();

    float child;


    // ------------------------------------------------------------
    // CROSSOVER
    // ------------------------------------------------------------

    if (random(1) < CROSSOVER_RATE) {

      float alpha =
        random(1);

      child =
        parentA +
        alpha * (parentB - parentA);

    } else {

      if (random(1) < 0.5) {

        child = parentA;

      } else {

        child = parentB;
      }
    }


    // ------------------------------------------------------------
    // MUTATION
    // ------------------------------------------------------------

    if (random(1) < MUTATION_RATE) {

      child +=
        randomGaussian() *
        MUTATION_SIGMA;
    }


    // ------------------------------------------------------------
    // BOUNDARY
    // ------------------------------------------------------------

    child =
      constrain(
        child,
        X_MIN,
        X_MAX
      );


    minNextPopulation[i] =
      child;
  }


  // --------------------------------------------------------------
  // REPLACE
  // --------------------------------------------------------------

  for (int i = 0; i < POP_SIZE; i++) {

    minPopulation[i] =
      minNextPopulation[i];
  }
}


// ================================================================
// EVOLVE MAXIMUM POPULATION
// ================================================================

void evolveMaximumPopulation() {

  // --------------------------------------------------------------
  // ELITISM
  // --------------------------------------------------------------

  int eliteIndex =
    findBestMaximumIndex();

  maxNextPopulation[0] =
    maxPopulation[eliteIndex];


  // --------------------------------------------------------------
  // CHILDREN
  // --------------------------------------------------------------

      maximumTournamentSelection();

    } else {

        child = parentA;

      } else {

        child = parentB;
      }
    }


    // ------------------------------------------------------------
    // MUTATION
    // ------------------------------------------------------------

    if (random(1) < MUTATION_RATE) {

      child +=

        MUTATION_SIGMA;
    }

    // BOUNDARY
    // ------------------------------------------------------------

  }


  // --------------------------------------------------------------

  }
}


// ================================================================
// MINIMUM TOURNAMENT
// ================================================================

float minimumTournamentSelection() {

  int bestIndex =
    int(random(POP_SIZE));


  for (int i = 1;
       i < TOURNAMENT_SIZE;
       i++) {

    int candidate =
      int(random(POP_SIZE));


    if (
      minFitness[candidate] <
      minFitness[bestIndex]
      ) {

      bestIndex =
        candidate;
    }
  }


  return minPopulation[bestIndex];
}


// ================================================================
// MAXIMUM TOURNAMENT
// ================================================================

float maximumTournamentSelection() {

  int bestIndex =
    int(random(POP_SIZE));


  for (int i = 1;
       i < TOURNAMENT_SIZE;
       i++) {

    int candidate =
      int(random(POP_SIZE));


    if (
      maxFitness[candidate] >
      maxFitness[bestIndex]
      ) {

      bestIndex =
        candidate;
    }
  }


  return maxPopulation[bestIndex];
}


// ================================================================
// FIND BEST MINIMUM
// ================================================================

int findBestMinimumIndex() {

  int index = 0;


  for (int i = 1;
       i < POP_SIZE;
       i++) {

    if (
      minFitness[i] <
      minFitness[index]
      ) {

      index = i;
    }
  }


  return index;
}


// ================================================================
// FIND BEST MAXIMUM
// ================================================================

int findBestMaximumIndex() {

  int index = 0;


  for (int i = 1;
       i < POP_SIZE;
       i++) {

    if (
      maxFitness[i] >
      maxFitness[index]
      ) {

      index = i;
    }
  }


  return index;
}

  ellipse(
    px,

    width * 0.009,

    );

    sx,
    sy + height * 0.120
    );


  // --------------------------------------------------------------
  // STATUS
  // --------------------------------------------------------------

  if (finished) {

    fill(150, 240, 170);

    textSize(width * 0.0095);


    text(
      "1000 ITERATIONS COMPLETED",
      sx,
      sy + height * 0.145
      );

  } else {

    fill(255, 205, 90);

    textSize(width * 0.0095);


    text(
      "EVOLUTION SEARCHING...",
      sx,
      sy + height * 0.145
      );
  }
}

void drawButtons() {

  float buttonY =
    height * 0.900;

  float buttonH =
    height * 0.060;

  float gap =
    width * 0.010;

  float buttonW =
    width * 0.145;

  float startX =
    width * 0.035;


  drawButton(
    startX,
    buttonY,
    buttonW,
    buttonH,
    "NEW RUN"
    );


  drawButton(
    startX + buttonW + gap,

    );

    startX + 2 * (buttonW + gap),
    buttonY,

    );


  drawButton(
    startX + 3 * (buttonW + gap),
    buttonY,
    buttonW,
    buttonH,
    "RESET"
    );
}

void drawButton(
  float x,
  float y,
  float w,
  float h,
  String label
  ) {

  fill(25, 43, 64);

  stroke(85, 125, 160);

  strokeWeight(2);


  rect(
    x,
    y,
    w,
    h,
    10
    );


  fill(240);

  textAlign(CENTER, CENTER);

  textSize(width * 0.010);


  text(
    label,
    x + w * 0.5,
    y + h * 0.5
    );


  textAlign(LEFT, TOP);
}


// ================================================================
// TOUCH
// ================================================================

void mousePressed() {

  float buttonY =
    height * 0.900;

  float buttonH =
    height * 0.060;

  float gap =
    width * 0.010;

  float buttonW =
    width * 0.145;

  float startX =
    width * 0.035;


  // --------------------------------------------------------------
  // NEW RUN
  // --------------------------------------------------------------

  if (
    inside(
      mouseX,
      mouseY,
      startX,
      buttonY,
      buttonW,
      buttonH
      )
    ) {

    resetGA();

    return;
  }


  // --------------------------------------------------------------
  // PAUSE / RUN
  // --------------------------------------------------------------

  if (
    inside(
      mouseX,
      mouseY,
      startX + buttonW + gap,
      buttonY,
      buttonW,
      buttonH
      )
    ) {

    if (!finished) {

      running =
        !running;


      if (running) {

        lastIterationTime =
          millis();
      }
    }

    return;
  }


  // --------------------------------------------------------------
  // STEP
  // --------------------------------------------------------------

  if (
    inside(
      mouseX,
      mouseY,
      startX + 2 * (buttonW + gap),
      buttonY,
      buttonW,
      buttonH
      )
    ) {

    if (!finished) {

      running = false;

      evolveOneIteration();
    }

    return;
  }


  // --------------------------------------------------------------
  // RESET
  // --------------------------------------------------------------

  if (
    inside(
      mouseX,
      mouseY,
      startX + 3 * (buttonW + gap),
      buttonY,
      buttonW,
      buttonH
      )
    ) {

    resetGA();

    return;
  }
}


// ================================================================
// HIT TEST
// ================================================================

boolean inside(
  float px,
  float py,
  float x,
  float y,
  float w,
  float h
  ) {

  return
    px >= x &&
    px <= x + w &&
    py >= y &&
    py <= y + h;
}
