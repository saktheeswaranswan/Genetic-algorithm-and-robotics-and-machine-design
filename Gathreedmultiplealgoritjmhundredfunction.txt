// ================================================================
// GAUSSIAN
// ================================================================

float G(
  float cx,
  float cy,
  float amp,
  float sigma
) {

  float r2 =
    sq(xPlaceholder(cx)) +
    sq(yPlaceholder(cy));

  return amp * exp(-r2 / (2.0*sigma*sigma));
}


// ================================================================
// HELPER FOR GAUSSIAN
// ================================================================

float currentXForGaussian;
float currentYForGaussian;

float xPlaceholder(float cx) {
  return currentXForGaussian - cx;
}

float yPlaceholder(float cy) {
  return currentYForGaussian - cy;
}


// ================================================================
// REAL FUNCTION EVALUATOR WRAPPER
// ================================================================

float functionValue(
  int id,
  float x,
  float y
) {

  currentXForGaussian = x;
  currentYForGaussian = y;

  return F(id,x,y);
}


// ================================================================
// SURFACE
// ================================================================

void generateSurface() {

  for (int i = 0; i < NX; i++) {

    float x =
      map(
        i,
        0,
        NX-1,
        XMIN,
        XMAX
      );

    for (int j = 0; j < NY; j++) {

      float y =
        map(
          j,
          0,
          NY-1,
          YMIN,
          YMAX
        );

      surfaceZ[i][j] =
        functionValue(
          functionID,
          x,
          y
        );
    }
  }
}


// ================================================================
// NUMERICAL DERIVATIVES
// ================================================================

float EPS = 0.015;


float fx(float x,float y) {

  return (
    functionValue(functionID,x+EPS,y)
    -
    functionValue(functionID,x-EPS,y)
  ) / (2.0*EPS);
}


float fy(float x,float y) {

  return (
    functionValue(functionID,x,y+EPS)
    -
    functionValue(functionID,x,y-EPS)
  ) / (2.0*EPS);
}


float fxx(float x,float y) {

  return (
    functionValue(functionID,x+EPS,y)
    -
    2.0*functionValue(functionID,x,y)
    +
    functionValue(functionID,x-EPS,y)
  ) / (EPS*EPS);
}


float fyy(float x,float y) {

  return (
    functionValue(functionID,x,y+EPS)
    -
    2.0*functionValue(functionID,x,y)
    +
    functionValue(functionID,x,y-EPS)
  ) / (EPS*EPS);
}


float fxy(float x,float y) {

  return (
    functionValue(functionID,x+EPS,y+EPS)
    -
    functionValue(functionID,x+EPS,y-EPS)
    -
    functionValue(functionID,x-EPS,y+EPS)
    +
    functionValue(functionID,x-EPS,y-EPS)
  ) / (4.0*EPS*EPS);
}


// ================================================================
// FITNESS
// ================================================================

float fitness(float x,float y) {

  float z =
    functionValue(
      functionID,
      x,
      y
    );

  if (searchMode == SEARCH_MAX) {
    return z;
  }

  if (searchMode == SEARCH_MIN) {
    return -z;
  }

  float gx = fx(x,y);
  float gy = fy(x,y);

  float a = fxx(x,y);
  float b = fyy(x,y);
  float c = fxy(x,y);

  float D =
    a*b - c*c;

  return
    1.0 /
    (
      0.01
      +
      abs(D)
      +
      0.6*sqrt(gx*gx+gy*gy)
    );
}


// ================================================================
// INITIAL POPULATION
// ================================================================

void initializePopulation() {

  for (int i=0;i<POPULATION_SIZE;i++) {

    population[i] =
      new Individual(
        random(XMIN,XMAX),
        random(YMIN,YMAX)
      );

    population[i].fitness =
      fitness(
        population[i].x,
        population[i].y
      );
  }
}


// ================================================================
// SORT
// ================================================================

void sortPopulation() {

  for(int i=0;i<POPULATION_SIZE-1;i++) {

    for(int j=i+1;j<POPULATION_SIZE;j++) {

      if(
        population[j].fitness >
        population[i].fitness
      ) {

        Individual t =
          population[i];

        population[i] =
          population[j];

        population[j] =
          t;
      }
    }
  }
}


// ================================================================
// TOURNAMENT
// ================================================================

Individual tournament() {

  Individual best = null;

  for(int k=0;k<5;k++) {

    int index =
      int(random(POPULATION_SIZE));

    Individual c =
      population[index];

    if(
      best == null ||
      c.fitness > best.fitness
    ) {

      best = c;
    }
  }

  return best;
}


// ================================================================
// GA GENERATION
// ================================================================

void runGeneration() {

  for(int i=0;i<POPULATION_SIZE;i++) {

    population[i].fitness =
      fitness(
        population[i].x,
        population[i].y
      );
  }

  sortPopulation();

  archiveCandidates();

  Individual[] next =
    new Individual[
      POPULATION_SIZE
    ];

  // Elite
  for(int i=0;i<ELITE_COUNT;i++) {

    next[i] =
      new Individual(
        population[i].x,
        population[i].y
      );
  }

  // Crossover and mutation
  for(
    int i=ELITE_COUNT;
    i<POPULATION_SIZE;
    i++
  ) {

    /*
       IMMIGRANT FRACTION
       prevents collapse onto
       one single optimum.
    */

    if(random(1)<0.25) {

      next[i] =
        new Individual(
          random(XMIN,XMAX),
          random(YMIN,YMAX)
        );

      continue;
    }

    Individual p1 =
      tournament();

    Individual p2 =
      tournament();

    float x =
      0.5*(p1.x+p2.x);

    float y =
      0.5*(p1.y+p2.y);

    if(random(1)<MUTATION_RATE) {

      x +=
        randomGaussian() *
        MUTATION_SIZE;
    }

    if(random(1)<MUTATION_RATE) {

      y +=
        randomGaussian() *
        MUTATION_SIZE;
    }

    next[i] =
      new Individual(
        constrain(x,XMIN,XMAX),
        constrain(y,YMIN,YMAX)
      );
  }

  population = next;

  generation++;

  if(generation%8==0) {

    verifyGrid();

    detectStationaryPoints();
  }
}


// ================================================================
// ARCHIVE CANDIDATES
// ================================================================

void archiveCandidates() {

  for(int i=0;i<POPULATION_SIZE;i++) {

    float x =
      population[i].x;

    float y =
      population[i].y;

    float z =
      functionValue(
        functionID,
        x,
        y
      );

    if(searchMode==SEARCH_MAX) {

      if(isLocalMaximumCandidate(x,y)) {

        addMaximum(x,y,z);
      }
    }

    else if(searchMode==SEARCH_MIN) {

      if(isLocalMinimumCandidate(x,y)) {

        addMinimum(x,y,z);
      }
    }

    else {

      if(isSaddleCandidate(x,y)) {

        addSaddle(x,y,z);
      }
    }
  }
}


// ================================================================
// LOCAL MAX CHECK
// ================================================================

boolean isLocalMaximumCandidate(
  float x,
  float y
) {

  float g =
    sqrt(
      fx(x,y)*fx(x,y)
      +
      fy(x,y)*fy(x,y)
    );

  float D =
    fxx(x,y)*fyy(x,y)
    -
    fxy(x,y)*fxy(x,y);

  return
    g < 0.35
    &&
    D > 0
    &&
    fxx(x,y) < 0;
}


// ================================================================
// LOCAL MIN CHECK
// ================================================================

boolean isLocalMinimumCandidate(
  float x,
  float y
) {

  float g =
    sqrt(
      fx(x,y)*fx(x,y)
      +
      fy(x,y)*fy(x,y)
    );

  float D =
    fxx(x,y)*fyy(x,y)
    -
    fxy(x,y)*fxy(x,y);

  return
    g < 0.35
    &&
    D > 0
    &&
    fxx(x,y) > 0;
}


// ================================================================
// SADDLE CHECK
// ================================================================

boolean isSaddleCandidate(
  float x,
  float y
) {

  float g =
    sqrt(
      fx(x,y)*fx(x,y)
      +
      fy(x,y)*fy(x,y)
    );

  float D =
    fxx(x,y)*fyy(x,y)
    -
    fxy(x,y)*fxy(x,y);

  return
    g < 0.35 &&
    D < 0;
}


// ================================================================
// ADD MAXIMUM
// ================================================================

void addMaximum(
  float x,
  float y,
  float z
) {

  if(
    duplicate(
      maxima,
      maxCount,
      x,y
    )
  ) {
    return;
  }

  if(maxCount<maxima.length) {

    maxima[maxCount] =
      new Feature(
        x,y,z
      );

    maxCount++;
  }
}

  ) {

  }

  }
}

  }

// ================================================================
// DUPLICATE
// ================================================================

boolean duplicate(
  Feature[] a,
  int count,
  float x,
  float y
) {

  for(int i=0;i<count;i++) {

    float dx =
      a[i].x-x;

    float dy =
      a[i].y-y;

    if(
      dx*dx+dy*dy
      <
      0.30*0.30
    ) {

      return true;
    }
  }

  return false;
}


// ================================================================
// GRID VERIFICATION
// ================================================================

void verifyGrid() {

  /*
     Independent grid verification.
     This is NOT replacing GA.

     It checks the entire domain numerically
     and catches boundary optima.
  */

  int N = 71;

  gridMaxZ = -1e30;
  gridMinZ = 1e30;

  for(int i=0;i<N;i++) {

    float x =
      map(
        i,
        0,
        N-1,
        XMIN,
        XMAX
      );

    for(int j=0;j<N;j++) {

      float y =
        map(
          j,
          0,
          N-1,
          YMIN,
          YMAX
        );

      float z =
        functionValue(
          functionID,
          x,
          y
        );

      if(z>gridMaxZ) {

        gridMaxZ=z;
        gridMaxX=x;
        gridMaxY=y;
      }

      if(z<gridMinZ) {

        gridMinZ=z;
        gridMinX=x;
        gridMinY=y;
      }
    }
  }

  /*
     Compare grid candidates with
     GA archive.
  */

  globalMaxZ = gridMaxZ;
  globalMaxX = gridMaxX;
  globalMaxY = gridMaxY;

  globalMinZ = gridMinZ;
  globalMinX = gridMinX;
  globalMinY = gridMinY;

  for(int i=0;i<maxCount;i++) {

    if(
      maxima[i].z >
      globalMaxZ
    ) {

      globalMaxZ =
        maxima[i].z;

      globalMaxX =
        maxima[i].x;

      globalMaxY =
        maxima[i].y;
    }
  }

  for(int i=0;i<minCount;i++) {

    if(
      minima[i].z <
      globalMinZ
    ) {

      globalMinZ =
        minima[i].z;

      globalMinX =
        minima[i].x;

      globalMinY =
        minima[i].y;
    }
  }
}


// ================================================================
// STATIONARY DETECTION
// ================================================================

void detectStationaryPoints() {

  int N = 55;

  for(int i=1;i<N-1;i++) {

    float x =
      map(
        i,
        0,
        N-1,
        XMIN,
        XMAX
      );

    for(int j=1;j<N-1;j++) {

      float y =
        map(
          j,
          0,
          N-1,
          YMIN,
          YMAX
        );

      if(isLocalMaximumCandidate(x,y)) {

        addMaximum(
          x,y,
          functionValue(
            functionID,x,y
          )
        );
      }

      if(isLocalMinimumCandidate(x,y)) {

        addMinimum(
          x,y,
          functionValue(
            functionID,x,y
          )
        );
      }

      if(isSaddleCandidate(x,y)) {

        addSaddle(
          x,y,
          functionValue(
            functionID,x,y
          )
        );
      }
    }
  }
}

// ================================================================
// SURFACE DRAW
// ================================================================

void drawSurface() {

  for(int i=0;i<NX-1;i++) {

    for(int j=0;j<NY-1;j++) {

      float x1 =
        map(i,0,NX-1,XMIN,XMAX);

      float x2 =
        map(i+1,0,NX-1,XMIN,XMAX);

      float y1 =
        map(j,0,NY-1,YMIN,YMAX);

      float y2 =
        map(j+1,0,NY-1,YMIN,YMAX);

      float z11 = surfaceZ[i][j];
      float z21 = surfaceZ[i+1][j];
      float z22 = surfaceZ[i+1][j+1];
      float z12 = surfaceZ[i][j+1];

      beginShape(TRIANGLES);

      noStroke();

      fill(surfaceColor(z11));

      vertex(
        map(x1,XMIN,XMAX,-260,260),
        map(y1,YMIN,YMAX,-260,260),
        z11*ZSCALE
      );

      vertex(
        map(x2,XMIN,XMAX,-260,260),
        map(y1,YMIN,YMAX,-260,260),
        z21*ZSCALE
      );

      vertex(
        map(x2,XMIN,XMAX,-260,260),
        map(y2,YMIN,YMAX,-260,260),
        z22*ZSCALE
      );

      fill(surfaceColor(z22));

      vertex(
        map(x1,XMIN,XMAX,-260,260),
        map(y1,YMIN,YMAX,-260,260),
        z11*ZSCALE
      );

      vertex(
        map(x2,XMIN,XMAX,-260,260),
        map(y2,YMIN,YMAX,-260,260),
        z22*ZSCALE
      );

      vertex(
        map(x1,XMIN,XMAX,-260,260),
        map(y2,YMIN,YMAX,-260,260),
        z12*ZSCALE
      );

      endShape();
    }
  }
}

// ================================================================
// BUTTON ACTION
// ================================================================

void buttonAction(
  float x,
  float y
) {

  if(
    x>700 && x<790 &&
    y>15 && y<53
  ) {

    searchMode=SEARCH_MAX;
    return;
  }

  if(
    x>800 && x<890 &&
    y>15 && y<53
  ) {

    searchMode=SEARCH_MIN;
    return;
  }

  if(
    x>900 && x<1010 &&
    y>15 && y<53
  ) {

    searchMode=SEARCH_SADDLE;
    return;
  }

  if(
    x>700 && x<790 &&
    y>60 && y<98
  ) {

    runGeneration();
    return;
  }

  if(
    x>800 && x<890 &&
    y>60 && y<98
  ) {

    resetSearch();
    return;
  }

  if(
    x>900 && x<1010 &&
    y>60 && y<98
  ) {

    autoRun=!autoRun;
    return;
  }

  if(
    x>1020 && x<1120 &&
    y>15 && y<53
  ) {

    functionID--;

    if(functionID<1)
      functionID=100;

    resetSearch();

    return;
  }

  if(
    x>1020 && x<1120 &&
    y>60 && y<98
  ) {

    functionID++;

    if(functionID>100)
      functionID=1;

    resetSearch();

    return;
  }
}


// ================================================================
// DRAG
// ================================================================

void mouseDragged() {

  float dx =
    mouseX-lastTouchX;

  float dy =
    mouseY-lastTouchY;

  rotY += dx*0.008;
  rotX += dy*0.008;

  rotX =
    constrain(
      rotX,
      -1.45,
      0.15
    );

  lastTouchX=mouseX;
  lastTouchY=mouseY;
}


// ================================================================
// RELEASE
// ================================================================

void mouseReleased() {

  dragging=false;
}


// ================================================================
// ANDROID MULTI-TOUCH
// ================================================================

public boolean surfaceTouchEvent(
  MotionEvent event
) {

  int action =
    event.getActionMasked();

  int count =
    event.getPointerCount();

  if(count>=2) {

    float x1 =
      event.getX(0);

    float y1 =
      event.getY(0);

    float x2 =
      event.getX(1);

    float y2 =
      event.getY(1);

    float d =
      dist(x1,y1,x2,y2);

    if(
      action==
      MotionEvent.ACTION_POINTER_DOWN
    ) {

      pinchDistance=d;
    }

    else if(
      action==
      MotionEvent.ACTION_MOVE
      &&
      pinchDistance>0
    ) {

      float change =
        d-pinchDistance;

      zoom *=
        1.0+change*0.002;

      zoom =
        constrain(
          zoom,
          0.45,
          2.5
        );

      pinchDistance=d;
    }
  }

  if(
    action==
    MotionEvent.ACTION_UP
    ||
    action==
    MotionEvent.ACTION_CANCEL
  ) {

    pinchDistance=-1;
  }

  return super.surfaceTouchEvent(event);
}

  verifyGrid();
}
